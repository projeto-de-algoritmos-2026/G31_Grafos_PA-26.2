# A versão é fixa de propósito. Com `spacy>=3.7` o pip podia resolver para
# o spaCy 3.7, que traz o thinc 8.2 — compilado contra o numpy 1.x — e
# instalar o numpy 2 ao lado. A combinação quebra na importação, com
# "ImportError: DLL load failed while importing numpy_ops". O spaCy 3.8
# traz o thinc 8.3, que já é compatível com o numpy 2.

numpy==2.2.6
